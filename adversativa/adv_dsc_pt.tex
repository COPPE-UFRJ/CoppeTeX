  \begin{algorithm}[H]
    \caption{Segundo algoritmo}\label{alg:adv2}
    \KwIn{Uma lista $L$}
    \KwOut{O maior elemento}
    $m \leftarrow -\infty$\;
    \ForEach{$x \in L$}{\lIf{$x > m$}{$m \leftarrow x$}}
    \Return $m$\;
  \end{algorithm}
  \source{Elaboração própria.}
  \begin{algorithm}[H]
    \caption{Terceiro algoritmo}\label{alg:adv3}
    \KwIn{$a$, $b$}
    \While{$b \neq 0$}{$(a,b) \leftarrow (b, a \bmod b)$\;}
    \Return $a$\;
  \end{algorithm}

  \begin{theglossary}
    \item[Adversativo] documento escrito para acionar ao mesmo tempo
      tudo o que a classe oferece, e assim revelar o que um exemplo
      bem-comportado esconde.
    \item[Fólio] o número da folha, impresso a 2\,cm da borda superior
      e da direita, em corpo menor (2.2(b) e 2.7).
    \item[Folha adicional] a folha que traz a ficha catalográfica e os
      campos da Coleta CAPES; não é contada nem numerada (3.1.2.1.2).
    \item[Gabarito] a referência como o Manual a imprime, contra a qual
      se compara a que a classe compôs.
    \item[Quadro] ilustração delimitada por linhas em todas as bordas,
      ao contrário da tabela, fechada só no topo e na base.
  \end{theglossary}
